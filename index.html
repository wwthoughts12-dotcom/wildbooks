async function loadData() {


    try {


        const [

            booksResponse,

            announcementsResponse,

            artsResponse

        ] = await Promise.all([

            fetch(BOOKS_URL),

            fetch(ANNOUNCEMENTS_URL),

            fetch(ARTS_URL)

        ]);


        books = await booksResponse.json();

        announcements = await announcementsResponse.json();

        arts = await artsResponse.json();


        updateStats();


        renderHome();


        renderCatalog();


        renderPlaylists();


        renderArts();


        document

            .getElementById('loading')

            .style.display = 'none';


    } catch(error){


        console.error(error);


        document.getElementById('loading').innerHTML =

        'Ошибка загрузки данных';


    }


}


function updateStats(){


    document.getElementById('booksCount').textContent =

    books.length;


    const uniqueSeries =

    [...new Set(

        books.map(

            b => b.series

        )

    )];


    document.getElementById('seriesCount').textContent =

    uniqueSeries.length;


    document.getElementById('playlistCount').textContent =

    books.filter(

        b => b.spotify_playlist

    ).length;


    let totalArts = 0;


    arts.forEach(art => {


        if(!art.arts) return;


        totalArts +=

        art.arts.split('|').length;


    });


    document.getElementById('artCount').textContent =

    totalArts;


}

function renderSeries(){

        'seriesContainer'

        seriesMap[book.series]++;

    container.innerHTML = '';


    Object.keys(seriesMap)

        .sort()

        .forEach(series => {


            container.innerHTML += `

function buildSeriesFilters(){


    const wrapper =

    document.getElementById(

        'seriesFilters'

    );


    const series =

    [

        'Все',

        ...new Set(

            books.map(

                b => b.series

            )

        )

    ];


    wrapper.innerHTML = '';


    series.forEach(item => {


        wrapper.innerHTML += `

        <button

            class="filter-btn ${

                item === currentSeries

                ? 'active'

                : ''

            }"

            onclick="selectSeries('${item}')">


            ${item}


        </button>

        `;


    });


}


function buildTropeFilters(){


    const wrapper =

    document.getElementById(

        'tropeFilters'

    );


    const tropeSet =

    new Set();


    books.forEach(book => {


        if(!book.trops) return;


        book.trops

            .split(',')

            .forEach(t => {


                tropeSet.add(

                    t.trim()

                );


            });


    });


    const tropes =

    ['Все', ...tropeSet];


    wrapper.innerHTML = '';


    tropes.forEach(item => {


        wrapper.innerHTML += `

        <button

            class="filter-btn ${

                item === currentTrope

                ? 'active'

                : ''

            }"

            onclick="selectTrope('${item}')">


            ${item}


        </button>

        `;


    });


}


function selectSeries(series){


    currentSeries = series;


    buildSeriesFilters();


    drawCatalog();


}


function selectTrope(trope){


    currentTrope = trope;


    buildTropeFilters();


    drawCatalog();


}


function drawCatalog(){


    const grid =

    document.getElementById(

        'catalogGrid'

    );


    const search =

    document

        .getElementById(

            'searchInput'

        )

        .value

        .toLowerCase();


    let filtered =

    [...books];


    if(currentSeries !== 'Все'){


        filtered =

        filtered.filter(

            book =>

            book.series === currentSeries

        );


    }


    if(currentTrope !== 'Все'){


        filtered =

        filtered.filter(

            book =>

            (book.trops || '')

            .includes(currentTrope)

        );


    }


    if(search){


        filtered =

        filtered.filter(book => {


            return (

                book.title

                    .toLowerCase()

                    .includes(search)


                ||


                book.author

                    .toLowerCase()

                    .includes(search)


                ||


                book.series

                    .toLowerCase()

                    .includes(search)

            );


        });


    }


    grid.innerHTML = '';


    filtered.forEach(book => {


        grid.innerHTML += `

        <div

            class="book-card"

            onclick="openBook('${book.ID}')">


            <img

                src="${book.cover_url}"

                class="book-cover"

            >


            <div class="book-info">


                <div class="book-title">

                    ${book.title}

                </div>


                <div class="book-author">

                    ${book.author}

                </div>


            </div>


        </div>

        `;


    });


}
